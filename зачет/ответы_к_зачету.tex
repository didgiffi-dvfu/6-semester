\section{2.2. Дифференцирование и интегрирование вектор-функций}

Пусть $V$ --- конечномерное евклидово векторное пространство,
$\vec{F} : (\alpha,\beta)\to V$.

\textbf{Определение производной.}
Вектор $\vec{a}$ называется производной функции $\vec{F}$ в точке $t_0$, если
\[
\lim_{t\to t_0}\frac{\vec{F}(t)-\vec{F}(t_0)}{t-t_0}=\vec{a}.
\]
Тогда пишут $\vec{F}'(t_0)=\vec{a}$.

\textbf{Покоординатное дифференцирование.}
Если в базисе $V$
\[
\vec{F}(t)=f^1(t)\vec{e}_1+\cdots+f^n(t)\vec{e}_n,
\]
то
\[
\vec{F}'(t_0)=\sum_{i=1}^n (f^i)'(t_0)\vec{e}_i.
\]
Следовательно, дифференцируемость вектор-функции эквивалентна
дифференцируемости всех её координатных функций.

\textbf{Основные правила дифференцирования.}
Если существуют соответствующие производные, то
\begin{align*}
(\vec{f}+\vec{g})' &= \vec{f}'+\vec{g}',\\
(c\vec{f})' &= c\vec{f}',\\
(\varphi \vec{f})' &= \varphi'\vec{f}+\varphi \vec{f}',\\
(\vec{f},\vec{g})' &= (\vec{f}',\vec{g})+(\vec{f},\vec{g}'),\\
[\vec{f},\vec{g}]' &= [\vec{f}',\vec{g}]+[\vec{f},\vec{g}'].
\end{align*}

\textbf{Интегрирование вектор-функций.}
Если
\[
\vec{f}(t)=f^1(t)\vec{e}_1+\cdots+f^n(t)\vec{e}_n
\]
непрерывна на $[\alpha,\beta]$, то интеграл определяется покоординатно:
\[
\int_\alpha^\beta \vec{f}(t)\dd t
=
\sum_{i=1}^n \left(\int_\alpha^\beta f^i(t)\dd t\right)\vec{e}_i.
\]

\textbf{Формула Ньютона--Лейбница.}
Если $\vec{F}'(t)=\vec{f}(t)$, то
\[
\int_\alpha^\beta \vec{f}(t)\dd t=\vec{F}(\beta)-\vec{F}(\alpha).
\]

\textbf{Оценка нормы интеграла.}
\[
\left\|\int_\alpha^\beta \vec{f}(t)\dd t\right\|
\le
\int_\alpha^\beta \|\vec{f}(t)\|\dd t.
\]

\textbf{Лемма о производной вектор-функции постоянного модуля.}
Если $\|\vec{F}(t)\|\equiv \mathrm{const}$, то
\[
(\vec{F}(t),\vec{F}(t))' = 2(\vec{F}(t),\vec{F}'(t))=0,
\]
следовательно,
\[
(\vec{F}(t),\vec{F}'(t))=0.
\]
То есть производная вектор-функции постоянного модуля ортогональна самой функции.

\section{2.3. Вектор скорости, касательная к кривой и нормальная плоскость}

\textbf{Параметризованная кривая.}
Непрерывное отображение
\[
\gamma:[\alpha,\beta]\to \R^3
\]
называется параметризованной кривой.

\textbf{Вектор скорости.}
Если $\gamma$ дифференцируема, то
\[
\gamma'(t_0)
\]
называется вектором скорости в момент $t_0$.
Если $\gamma'(t_0)\neq 0$, то кривая в точке $\gamma(t_0)$ регулярна.

\textbf{Касательная прямая.}
Если
\[
\gamma(t)=(x(t),y(t),z(t)),\qquad \gamma'(t_0)\neq 0,
\]
то касательная прямая в точке $A=\gamma(t_0)$ имеет уравнение
\[
\vect{r}=\gamma(t_0)+\lambda \gamma'(t_0),
\qquad \lambda\in \R,
\]
то есть
\[
(X,Y,Z)=(x(t_0),y(t_0),z(t_0))+\lambda(x'(t_0),y'(t_0),z'(t_0)).
\]

\textbf{Нормальная плоскость.}
Нормальной плоскостью к регулярной кривой в точке $A=\gamma(t_0)$ называется
плоскость, проходящая через $A$ и ортогональная касательному вектору
$\gamma'(t_0)$. Её уравнение:
\[
(\vect{r}-\gamma(t_0),\gamma'(t_0))=0.
\]
Если
\[
\gamma'(t_0)=(x'(t_0),y'(t_0),z'(t_0)),
\]
то
\[
x'(t_0)\bigl(X-x(t_0)\bigr)
+y'(t_0)\bigl(Y-y(t_0)\bigr)
+z'(t_0)\bigl(Z-z(t_0)\bigr)=0.
\]

\section{Длина кривой и естественная параметризация}

Пусть $\gamma:[\alpha,\beta]\to \R^3$ --- регулярная $C^1$-кривая.

\textbf{Длина кривой.}
\[
L(\gamma)=\int_\alpha^\beta \|\gamma'(t)\|\dd t.
\]
В координатах:
\[
L(\gamma)=\int_\alpha^\beta
\sqrt{(x'(t))^2+(y'(t))^2+(z'(t))^2}\dd t.
\]

\textbf{Естественный параметр.}
Функция
\[
s=s(t)=\int_{t_0}^t \|\gamma'(\tau)\|\dd \tau
\]
называется длиной дуги, отсчитанной от точки $t_0$.

Так как $\|\gamma'(t)\|>0$, функция $s(t)$ строго монотонна и локально обратима.
После перепараметризации $\tilde{\gamma}(s)=\gamma(t(s))$ получаем
\[
\left\|\frac{\dd\tilde{\gamma}}{\dd s}\right\|=1.
\]
Такая параметризация называется \textbf{естественной}.

\textbf{Смысл естественной параметризации.}
Если точки $A=\tilde{\gamma}(s_1)$ и $B=\tilde{\gamma}(s_2)$ лежат на кривой,
то длина дуги между ними равна
\[
|s_2-s_1|.
\]

\section{Кривизна, кручение и формулы Френе}

Пусть $\gamma=\gamma(s)$ --- регулярная кривая класса $C^3$, заданная
естественным параметром $s$.

\textbf{Единичный касательный вектор:}
\[
\vec{\tau}(s)=\gamma'(s),\qquad \|\vec{\tau}(s)\|=1.
\]

\textbf{Кривизна:}
\[
\kappa(s)=\|\vec{\tau}'(s)\|.
\]
Если $\kappa(s)\neq 0$, то единичная главная нормаль определяется формулой
\[
\vec{n}(s)=\frac{\vec{\tau}'(s)}{\kappa(s)}.
\]

\textbf{Бинормаль:}
\[
\vec{b}(s)=\vec{\tau}(s)\times \vec{n}(s).
\]

\textbf{Кручение} $\varkappa(s)$ определяется из формулы Френе
\[
\vec{b}'(s)=-\varkappa(s)\vec{n}(s).
\]

\textbf{Формулы Френе:}
\begin{align*}
\vec{\tau}'(s) &= \kappa(s)\vec{n}(s),\\
\vec{n}'(s) &= -\kappa(s)\vec{\tau}(s)+\varkappa(s)\vec{b}(s),\\
\vec{b}'(s) &= -\varkappa(s)\vec{n}(s).
\end{align*}

\textbf{Формулы для произвольного параметра $t$.}
Если $\gamma'(t)\neq 0$, то
\[
\kappa(t)=\frac{\|\gamma'(t)\times\gamma''(t)\|}{\|\gamma'(t)\|^3},
\]
а при $\gamma'(t)\times\gamma''(t)\neq 0$
\[
\varkappa(t)=
\frac{\det(\gamma'(t),\gamma''(t),\gamma'''(t))}
{\|\gamma'(t)\times\gamma''(t)\|^2}.
\]

\section{Условия равенства нулю кривизны и кручения}

Пусть $\gamma$ --- регулярная кривая.

\textbf{1. Условие $\kappa\equiv 0$.}
Если
\[
\kappa(s)=\|\vec{\tau}'(s)\|\equiv 0,
\]
то
\[
\vec{\tau}'(s)=0,
\]
следовательно, $\vec{\tau}(s)$ постоянно.
Значит, все касательные совпадают, и кривая лежит на одной прямой.

\textbf{2. Условие $\varkappa\equiv 0$.}
Пусть $\gamma$ класса $C^3$ и $\kappa>0$ на рассматриваемом промежутке.
Если
\[
\varkappa(s)\equiv 0,
\]
то по формуле Френе
\[
\vec{b}'(s)=0,
\]
следовательно, бинормаль $\vec{b}$ постоянна.
Тогда скалярное произведение
\[
(\gamma(s)-\gamma(s_0),\vec{b})
\]
постоянно, а значит, кривая лежит в плоскости, ортогональной постоянному вектору $\vec{b}$.

\section{Параметризация, регулярная параметризация и поверхность в евклидовом пространстве}

\textbf{Параметризация поверхности.}
Отображение
\[
r:U\to \R^3,
\qquad U\subset \R^2 \text{ открыто},
\]
называется параметризацией, если его образ задаёт участок поверхности.

\textbf{Регулярная параметризация.}
Параметризация $r(u,v)$ называется регулярной, если
\begin{itemize}
  \item $r$ по крайней мере класса $C^1$;
  \item частные производные $r_u$ и $r_v$ существуют и непрерывны;
  \item
  \[
  r_u(u,v)\times r_v(u,v)\neq 0
  \qquad \forall (u,v)\in U.
  \]
\end{itemize}
Последнее условие равносильно тому, что дифференциал $\dd r$ имеет ранг $2$.

\textbf{Поверхность.}
Множество $S\subset \R^3$ называется регулярной поверхностью, если для каждой
точки $p\in S$ существует окрестность $W\subset \R^3$ и регулярная параметризация
\[
r:U\to S\cap W,
\]
которая является гомеоморфизмом $U$ на свой образ.

\section{График функции двух переменных как поверхность. Поверхности вращения}

\textbf{График функции как поверхность.}
Если $f:U\to \R$ --- функция класса $C^1$, где $U\subset \R^2$ открыто, то её график
\[
\Gamma_f=\{(u,v,f(u,v))\mid (u,v)\in U\}
\]
является регулярной поверхностью.

Действительно, параметризация графика имеет вид
\[
r(u,v)=(u,v,f(u,v)),
\]
причём
\[
r_u=(1,0,f_u),\qquad r_v=(0,1,f_v),
\]
и
\[
r_u\times r_v=(-f_u,-f_v,1)\neq 0.
\]

\textbf{Замечание о корректности.}
В черновике было написано, что достаточно непрерывности $f$.
Для \textbf{гладкой} поверхности этого недостаточно: нужна хотя бы $C^1$-гладкость.

\textbf{Поверхность вращения.}
Пусть в плоскости $Oxz$ задана кривая
\[
\gamma(u)=(f(u),0,g(u)),
\]
где $f(u)>0$ и
\[
(f'(u))^2+(g'(u))^2\neq 0.
\]
При вращении вокруг оси $Oz$ получаем параметризацию
\[
r(u,v)=\bigl(f(u)\cos v,\ f(u)\sin v,\ g(u)\bigr),
\qquad v\in [0,2\pi).
\]

\section{Первая и вторая основные формы поверхности}

Пусть $S$ --- регулярная поверхность, $p\in S$, $T_pS$ --- касательная плоскость.

\textbf{Первая основная форма.}
Это ограничение евклидова скалярного произведения на касательную плоскость:
\[
I_p(\vec{a},\vec{b})=(\vec{a},\vec{b}),\qquad \vec{a},\vec{b}\in T_pS.
\]

Если $r=r(u,v)$ --- регулярная параметризация, то
\[
E=(r_u,r_u),\qquad F=(r_u,r_v),\qquad G=(r_v,r_v),
\]
и
\[
I=E\dd u^2+2F\dd u\dd v+G\dd v^2.
\]

\textbf{Вторая основная форма.}
Выберем единичную нормаль
\[
\vec{m}=\frac{r_u\times r_v}{\|r_u\times r_v\|}.
\]
Тогда
\[
L=(r_{uu},\vec{m}),\qquad M=(r_{uv},\vec{m}),\qquad N=(r_{vv},\vec{m}),
\]
и
\[
II=L\dd u^2+2M\dd u\dd v+N\dd v^2.
\]

\textbf{Геометрический смысл.}
Если кривая на поверхности задаётся как $r(u(t),v(t))$, то нормальная
составляющая ускорения равна
\[
\kappa_n\,\|\gamma'(t)\|^2
=
L(u')^2+2Mu'v'+N(v')^2.
\]

\textbf{Зависимость от ориентации.}
При замене нормали $\vec{m}\mapsto -\vec{m}$ коэффициенты $L,M,N$ меняют знак,
поэтому вторая основная форма определена с точностью до знака.

\section{Нормальная кривизна. Теорема Мёнье}

Пусть $p\in S$, а $\vec{\xi}\in T_pS$, $\vec{\xi}\neq 0$.

\textbf{Нормальная кривизна} в направлении $\vec{\xi}$ определяется формулой
\[
\kappa_n(\vec{\xi})=\frac{II_p(\vec{\xi},\vec{\xi})}{I_p(\vec{\xi},\vec{\xi})}.
\]
Если
\[
\vec{\xi}=u' r_u+v' r_v,
\]
то
\[
\kappa_n=
\frac{L(u')^2+2Mu'v'+N(v')^2}{E(u')^2+2Fu'v'+G(v')^2}.
\]

\textbf{Теорема Мёнье.}
Пусть две кривые на поверхности проходят через одну и ту же точку
и имеют там одно и то же касательное направление.
Тогда величина нормальной кривизны зависит только от этого направления,
а не от выбора кривой.

Если $\kappa$ --- кривизна кривой как пространственной кривой, а $\phi$ ---
угол между её главной нормалью и нормалью поверхности, то
\[
\kappa_n=\kappa\cos\phi.
\]
Отсюда видно, что среди всех кривых поверхности с фиксированным касательным
направлением нормальная кривизна одна и та же.

\section{Главные кривизны, главные направления. Теорема Эйлера}

В точке $p$ нормальная кривизна есть квадратичная форма от направления в касательной плоскости.
Рассматривая оператор формы Вейнгартена, получают два собственных значения
$k_1$ и $k_2$, называемых \textbf{главными кривизнами}, и соответствующие
собственные направления, называемые \textbf{главными направлениями}.

\textbf{Свойства.}
\begin{itemize}
  \item Главные направления взаимно ортогональны
  (если $k_1\neq k_2$).
  \item Главные кривизны --- это экстремальные значения нормальной кривизны в точке.
\end{itemize}

\textbf{Теорема Эйлера.}
Если направление $\vec{\xi}$ образует угол $\theta$ с главным направлением,
соответствующим кривизне $k_1$, то
\[
\kappa_n(\theta)=k_1\cos^2\theta+k_2\sin^2\theta.
\]

\textbf{Средняя и гауссова кривизны:}
\[
H=\frac{k_1+k_2}{2},
\qquad
K=k_1k_2.
\]

\section{Уравнение для главных кривизн. Гауссова кривизна}

Пусть
\[
I=
\begin{pmatrix}
E & F\\
F & G
\end{pmatrix},
\qquad
II=
\begin{pmatrix}
L & M\\
M & N
\end{pmatrix}.
\]

\textbf{Уравнение для главных кривизн.}
Главные кривизны являются корнями характеристического уравнения
\[
\det(II-kI)=0,
\]
то есть
\[
\det
\begin{pmatrix}
L-kE & M-kF\\
M-kF & N-kG
\end{pmatrix}
=0.
\]
После раскрытия determinant'а получаем
\[
(EG-F^2)k^2-(EN-2FM+GL)k+(LN-M^2)=0.
\]

\textbf{Гауссова кривизна:}
\[
K=\frac{LN-M^2}{EG-F^2}=k_1k_2.
\]

\textbf{Средняя кривизна:}
\[
H=\frac{EN-2FM+GL}{2(EG-F^2)}=\frac{k_1+k_2}{2}.
\]

\textbf{Теорема Гаусса (Theorema Egregium).}
Гауссова кривизна $K$ выражается только через коэффициенты первой основной формы
$E,F,G$ и их производные. Поэтому $K$ --- внутренняя характеристика поверхности.

\section{Деривационные уравнения. Символы Кристоффеля}

Пусть $r=r(u,v)$ --- регулярная параметризация поверхности.
Так как $r_{uu},r_{uv},r_{vv}$ лежат в линейной оболочке $r_u,r_v,\vec{m}$,
их можно разложить по этому базису:
\begin{align*}
r_{uu} &= \Gamma_{11}^1 r_u+\Gamma_{11}^2 r_v+L\vec{m},\\
r_{uv} &= \Gamma_{12}^1 r_u+\Gamma_{12}^2 r_v+M\vec{m},\\
r_{vv} &= \Gamma_{22}^1 r_u+\Gamma_{22}^2 r_v+N\vec{m}.
\end{align*}
Это \textbf{уравнения Гаусса}.

Для производных нормали получаем \textbf{уравнения Вейнгартена}:
\begin{align*}
\vec{m}_u &= -b_1^1 r_u-b_1^2 r_v,\\
\vec{m}_v &= -b_2^1 r_u-b_2^2 r_v,
\end{align*}
где
\[
b_i^j=g^{jk}b_{ik}.
\]

\textbf{Символы Кристоффеля второго рода} $\Gamma^k_{ij}$ ---
это коэффициенты касательной части вторых производных параметризации, то есть
\[
\nabla_{\partial_i}\partial_j=\Gamma^k_{ij}\partial_k.
\]

\textbf{Символы первого рода} определяются по формуле
\[
\Gamma_{k,ij}=g_{k\ell}\Gamma^\ell_{ij}
=
\frac12\left(
\frac{\partial g_{ik}}{\partial u^j}
+
\frac{\partial g_{jk}}{\partial u^i}
-
\frac{\partial g_{ij}}{\partial u^k}
\right).
\]

\textbf{Через метрический тензор:}
\[
\Gamma^k_{ij}
=
\frac12 g^{k\ell}
\left(
\frac{\partial g_{i\ell}}{\partial u^j}
+
\frac{\partial g_{j\ell}}{\partial u^i}
-
\frac{\partial g_{ij}}{\partial u^\ell}
\right).
\]

\section{Уравнения Гаусса--Кодацци}

Деривационные уравнения должны быть совместны, потому что
\[
r_{uv}=r_{vu},
\qquad
\vec{m}_{uv}=\vec{m}_{vu}.
\]
Из этого следуют \textbf{уравнения Гаусса--Кодацци}.

\textbf{Координатная форма уравнений Кодацци:}
\[
b_{ij;k}=b_{ik;j},
\]
где ``$;$'' означает ковариантное дифференцирование по первой основной форме.

В частности,
\begin{align*}
L_v-M_u &= L\Gamma_{12}^1+M(\Gamma_{12}^2-\Gamma_{11}^1)-N\Gamma_{11}^2,\\
M_v-N_u &= L\Gamma_{22}^1+M(\Gamma_{22}^2-\Gamma_{12}^1)-N\Gamma_{12}^2.
\end{align*}

\textbf{Уравнение Гаусса:}
\[
LN-M^2=K(EG-F^2).
\]

В матричной записи условие совместности можно записать так:
\[
\frac{\partial A_2}{\partial u}-\frac{\partial A_1}{\partial v}+[A_1,A_2]=0.
\]

\section{Ковариантное дифференцирование. Уравнения геодезических. Геодезические на поверхностях вращения}

\textbf{Ковариантное дифференцирование вдоль кривой.}
Пусть на поверхности задана кривая
\[
\gamma(t)=r(u^1(t),u^2(t))
\]
и касательное векторное поле вдоль неё
\[
\vec{\xi}(t)=\xi^1(t)r_1+\xi^2(t)r_2.
\]
Тогда ковариантная производная поля $\vec{\xi}$ вдоль $\gamma$ определяется как
\[
\frac{D\vec{\xi}}{dt}
=
\left(
\frac{\dd \xi^k}{\dd t}
+
\Gamma^k_{ij}\xi^i\frac{\dd u^j}{\dd t}
\right)r_k.
\]

\textbf{Ковариантная производная касательного вектора.}
Если
\[
\dot{\gamma}
=
\frac{\dd u^k}{\dd t}r_k,
\]
то
\[
\frac{D\dot{\gamma}}{dt}
=
\left(
\frac{\dd^2u^k}{\dd t^2}
+
\Gamma^k_{ij}\frac{\dd u^i}{\dd t}\frac{\dd u^j}{\dd t}
\right)r_k.
\]

\textbf{Геодезическая.}
Геодезическая линия --- это кривая на поверхности, у которой геодезическая
кривизна равна нулю. Эквивалентно,
\[
\frac{D\dot{\gamma}}{dt}=0
\]
при аффинном параметре, а в частности при параметре длины дуги.

\textbf{Уравнения геодезических в натуральном параметре $s$:}
\[
\frac{\dd^2u^k}{\dd s^2}
+
\Gamma^k_{ij}\frac{\dd u^i}{\dd s}\frac{\dd u^j}{\dd s}
=0,
\qquad k=1,2.
\]
То есть
\begin{align*}
\frac{\dd^2u^1}{\dd s^2}
+\Gamma^1_{ij}\frac{\dd u^i}{\dd s}\frac{\dd u^j}{\dd s}&=0,\\
\frac{\dd^2u^2}{\dd s^2}
+\Gamma^2_{ij}\frac{\dd u^i}{\dd s}\frac{\dd u^j}{\dd s}&=0.
\end{align*}

\textbf{Существование и единственность.}
Через каждую точку поверхности и каждое касательное направление проходит
единственная геодезическая.

\textbf{Геодезические на поверхности вращения.}
Для поверхности вращения выполняется \textbf{соотношение Клеро}.
Если $\rho$ --- расстояние от точки до оси вращения, а $\alpha$ --- угол
между геодезической и параллелью, то
\[
\rho\cos\alpha=\mathrm{const}.
\]
Если $\beta$ --- угол между геодезической и меридианом, то эквивалентно
\[
\rho\sin\beta=\mathrm{const}.
\]
В частности, меридианы всегда являются геодезическими.

\section{Геодезические как экстремали функционалов действия и длины}

Пусть кривая на поверхности задана в координатах
\[
t\mapsto (u^1(t),u^2(t)).
\]

\textbf{Функционал длины:}
\[
\mathcal{L}[\gamma]
=
\int_a^b
\sqrt{g_{ij}(u(t))\dot{u}^i(t)\dot{u}^j(t)}\dd t.
\]
Он равен длине кривой на поверхности.

\textbf{Функционал действия (энергии):}
\[
\mathcal{E}[\gamma]
=
\frac12\int_a^b g_{ij}(u(t))\dot{u}^i(t)\dot{u}^j(t)\dd t.
\]

\textbf{Связь между ними.}
Экстремали функционала длины и функционала энергии совпадают после
перехода к натуральной параметризации. Поэтому геодезические можно
получать как экстремали любого из этих двух функционалов.

\textbf{Уравнения Эйлера--Лагранжа.}
Для лагранжиана
\[
L(u,\dot{u})=\frac12 g_{ij}(u)\dot{u}^i\dot{u}^j
\]
условия Эйлера--Лагранжа
\[
\frac{\dd}{\dd t}\frac{\partial L}{\partial \dot{u}^k}
-\frac{\partial L}{\partial u^k}=0
\]
приводят к уравнениям
\[
\ddot{u}^k+\Gamma^k_{ij}\dot{u}^i\dot{u}^j=0.
\]
Это и есть уравнения геодезических.

\textbf{Локальное кратчайшее свойство.}
Достаточно малый отрезок геодезической даёт кратчайший путь между своими концами
среди всех близких кривых на поверхности, соединяющих те же точки.
